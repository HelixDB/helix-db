# SQL → Cypher translations, ported from neo4j-jdbc's translator tests
# (neo4j-jdbc-translator/impl/src/test, Apache-2.0) and extended.
#
# Each block: `#` comments, optional `-- tables: a=B, c=D` and
# `-- joins: t.col=TYPE` mappings, the SQL, then `=>` and either the expected
# Cypher or `error: <category> <detail>`. Expected Cypher compares after
# parsing and printing, so spacing and redundant parentheses do not matter.
# A trailing `!! <category> <detail>` expects HelixDB to reject the translated
# Cypher when it compiles; otherwise it must compile.
# `divergence:` comments mark where HelixDB's translation differs from
# neo4j-jdbc's and why. Throughout, SUM(x) becomes
# `CASE count(x) WHEN 0 THEN NULL ELSE sum(x) END`: SQL's SUM of no values is
# null, Cypher's sum is 0. A window (ORDER BY with SKIP or LIMIT) without
# DISTINCT or grouping applies to the table variables in a WITH before RETURN
# reads columns, so HelixDB can stop a scan at the LIMIT.

# predicates.adoc p1_2
# divergence: `(a, b) <> (c, d)` holds when any field differs (OR), as in SQL;
# neo4j-jdbc requires every field to differ.
SELECT 1
FROM p
WHERE (1, 2) = (3, 4)
OR (1, 2) < (3, 4)
OR (1, 2) <= (3, 4)
OR (1, 2, 3) <> (4, 5, 6)
OR (1, 2, 3) > (4, 5, 6)
OR (1, 2, 3) >= (4, 5, 6)
=> MATCH (p:p)
WHERE 1 = 3 AND 2 = 4
OR (1 < 3 OR 1 = 3 AND 2 < 4)
OR (1 < 3 OR 1 = 3 AND 2 <= 4)
OR (1 <> 4 OR 2 <> 5 OR 3 <> 6)
OR (1 > 4 OR 1 = 4 AND (2 > 5 OR 2 = 5 AND 3 > 6))
OR (1 > 4 OR 1 = 4 AND (2 > 5 OR 2 = 5 AND 3 >= 6))
RETURN 1

# Named parameters with an at sign, as in SQL Server; a quoted name is a column

=> MATCH (m:Movie) WHERE m.released > $year AND m.`@title` = $title RETURN m.title

=> CREATE (movie:Movie {title: $title})
